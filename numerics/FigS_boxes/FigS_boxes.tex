\newcommand{\gridtable}[1]{../Fig3_grid_update/plotting_tables/fig3_K#1_summary.csv}
\pgfplotsset{
  selected field/.style={x filter/.expression={abs(\thisrow{E}-#1)<1e-6 ? x : nan}},
  curve/.style={thick,mark size=1.7pt},
  field curve/.style={curve,forget plot},
  rate error/.style={error bars/y dir=both,error bars/y explicit,error bars/error mark options={mark size=1.2pt}}
}
\newcommand{\ratepanel}[3]{%
  \nextgroupplot[title={$\boldsymbol{#1\times #1}$},ymin=0,ymax=0.85,
    ytick={0,.2,.4,.6,.8},ylabel={#3},xticklabels=\empty]
  \addplot[curve,cblue,mark=*,selected field=0.1,rate error]
    table[col sep=comma,x=N,y=true_rate,y error=true_rate_sd]{\gridtable{#1}};
  \addplot[curve,cteal,mark=square*,dashed,selected field=0.1,rate error]
    table[col sep=comma,x=N,y=estimator_rate,y error=estimator_rate_sd]{\gridtable{#1}};
  \node[anchor=north west,inner sep=2.5pt] at (rel axis cs:0,1) {#2};
}
\newcommand{\fieldseries}[3]{%
  \addplot[field curve,selected field=#2,#3]
    table[col sep=comma,x=N,y=quality_factor_Q]{\gridtable{#1}};
}
\newcommand{\qualitypanel}[3]{%
  \nextgroupplot[ymin=0,ymax=.82,ytick={0,.2,.4,.6,.8},ylabel={#3},xlabel={$N$}]
  \addplot[black!60,densely dotted,thick] coordinates {(-3,0.666666667)(104,0.666666667)};
  \fieldseries{#1}{0.1}{cblue,mark=*}
  \fieldseries{#1}{0.2}{cteal,mark=square*,dashed}
  \fieldseries{#1}{0.4}{cgreen,mark=triangle*,dashdotted}
  \fieldseries{#1}{0.5}{corange,mark=diamond*,densely dotted}
  \fieldseries{#1}{0.8}{cpurple,mark=triangle*,mark options={rotate=180},dashed}
  \fieldseries{#1}{1.0}{cbrown,mark=pentagon*,dashdotted}
  \node[anchor=north west,inner sep=2.5pt] at (rel axis cs:0,1) {#2};
}

}
